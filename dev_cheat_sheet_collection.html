SELECT Basics

SELECT * FROM users;                 -- all columns
SELECT name, email FROM users;       -- specific columns
SELECT DISTINCT country FROM users;  -- unique values
SELECT name AS full_name FROM users; -- alias

SELECT * FROM users
WHERE age >= 18
ORDER BY created_at DESC              -- DESC / ASC
LIMIT 10 OFFSET 20;                  -- pagination

Filtering (WHERE)

WHERE age = 18
WHERE age <> 18              -- not equal (also !=)
WHERE age BETWEEN 18 AND 30
WHERE country IN ('US', 'CA', 'UK')
WHERE name LIKE 'A%'         -- % = any chars, _ = one char
WHERE email IS NULL
WHERE email IS NOT NULL
WHERE age > 18 AND (country = 'US' OR vip = TRUE)

Joins

Combine rows from multiple tables on a related column.

-- INNER: only rows matching in both tables
SELECT o.id, u.name
FROM orders o
JOIN users u ON o.user_id = u.id;

-- LEFT: all left rows; NULLs where no match on the right
SELECT u.name, o.id
FROM users u
LEFT JOIN orders o ON o.user_id = u.id;

-- RIGHT / FULL OUTER: mirror image / both sides
-- CROSS JOIN: every combination (cartesian product)

-- self-join (a table joined to itself, use aliases)
SELECT e.name, m.name AS manager
FROM employees e
JOIN employees m ON e.manager_id = m.id;

Aggregation (GROUP BY)

SELECT COUNT(*) FROM users;
SELECT AVG(age), MIN(age), MAX(age), SUM(balance) FROM users;

SELECT country, COUNT(*) AS n
FROM users
WHERE active = TRUE          -- WHERE filters ROWS (before grouping)
GROUP BY country
HAVING COUNT(*) > 100        -- HAVING filters GROUPS (after)
ORDER BY n DESC;

-- clause order:
-- SELECT .. FROM .. WHERE .. GROUP BY .. HAVING .. ORDER BY .. LIMIT

Subqueries & CTEs

-- subquery in WHERE
SELECT name FROM users
WHERE id IN (SELECT user_id FROM orders WHERE total > 100);

-- correlated subquery
SELECT name FROM users u
WHERE EXISTS (SELECT 1 FROM orders o WHERE o.user_id = u.id);

-- CTE (Common Table Expression): readable + reusable
WITH big_spenders AS (
    SELECT user_id, SUM(total) AS spent
    FROM orders
    GROUP BY user_id
    HAVING SUM(total) > 1000
)
SELECT u.name, b.spent
FROM big_spenders b
JOIN users u ON u.id = b.user_id;

Window Functions

SELECT
  name, department, salary,
  ROW_NUMBER() OVER (PARTITION BY department
                     ORDER BY salary DESC) AS rn,
  RANK()       OVER (ORDER BY salary DESC) AS rnk,
  AVG(salary)  OVER (PARTITION BY department) AS dept_avg
FROM employees;

-- running total
SUM(amount) OVER (ORDER BY day
  ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)

-- previous / next row
LAG(salary)  OVER (ORDER BY hire_date)
LEAD(salary) OVER (ORDER BY hire_date)

git remote add origin <url>
git fetch                # download, no merge
git pull                 # fetch + merge
git push                 # push commits
git push -u origin main  # set upstream (first push)
git push --force-with-lease   # safer forced push

Inspect History

git log
git log --oneline --graph --all   # compact visual graph
git log -p file          # history with diffs
git diff                 # unstaged changes
git diff --staged        # staged changes
git show <commit>        # one commit in detail
git blame file           # who changed each line

Undoing Things

git restore file              # discard unstaged changes
git restore --staged file     # unstage
git reset --soft HEAD~1       # undo commit, keep staged
git reset --mixed HEAD~1      # undo commit, unstage (default)
git reset --hard HEAD~1       # discard commit AND changes (!)
git revert <commit>           # new commit that undoes one (safe)

Stashing

git stash                # shelve uncommitted work
git stash -u             # include untracked
git stash list
git stash pop            # reapply + remove
git stash apply          # reapply, keep in list
git stash drop

Rebasing

git rebase main          # replay current branch onto main
git rebase -i HEAD~3     # interactive: squash/reword/reorder
git rebase --continue    # after resolving conflicts
git rebase --abort       # bail out
# Rule: don't rebase commits you've already pushed/shared

Tags

git tag                  # list
git tag v1.0.0           # lightweight
git tag -a v1.0.0 -m "release"    # annotated
git push origin v1.0.0   # push one tag
git push origin --tags   # push all tags

Cherry-pick & Rescue

git cherry-pick <commit> # apply a single commit here
git reflog               # where HEAD has been (recover lost work)
git bisect start         # binary-search a bad commit
git clean -fd            # delete untracked files & dirs (!)
git mv old new ; git rm file

dist/
.DS_Store

Images

docker pull nginx ; docker pull nginx:1.27
docker images                 # list local images
docker build -t myapp:1.0 .   # build from Dockerfile
docker tag myapp:1.0 user/myapp:1.0
docker rmi nginx              # remove image
docker image prune            # remove dangling

Containers

docker run -d -p 8080:80 --name web nginx
docker run -it ubuntu bash    # interactive shell
docker run --rm alpine echo hi    # auto-remove on exit
docker run -e KEY=value app   # env var

docker ps ; docker ps -a      # running / all
docker stop <id> ; docker start <id> ; docker restart <id>
docker rm <id> ; docker rm -f <id>

Exec, Logs & Inspect

docker exec -it <id> bash     # shell into running container
docker logs -f <id>           # follow logs
docker inspect <id>           # full JSON details
docker stats                  # live resource usage
docker cp <id>:/path ./local  # copy out

Dockerfile

# recipe to build an image
FROM node:20-alpine
WORKDIR /app
COPY package*.json ./
RUN npm install
COPY . .
EXPOSE 3000
CMD ["node", "server.js"]
# build:  docker build -t myapp .

Volumes & Networks

docker volume create data
docker run -v data:/var/lib app     # named volume
docker run -v $(pwd):/app app       # bind-mount cwd
docker volume ls

docker network create mynet
docker run --network mynet app
docker network ls

Docker Compose

# docker-compose.yml defines multi-container apps
docker compose up -d          # start (detached)
docker compose down           # stop & remove
docker compose ps
docker compose logs -f        # follow logs
docker compose build
docker compose exec web bash

Registry & Cleanup

docker login
docker push user/myapp:1.0
docker system prune           # remove unused data
docker system prune -a        # also unused images
docker system df              # disk usage

K8s · kubectl Basics

Kubernetes orchestrates containers across a cluster. kubectl is the CLI.

kubectl version --client
kubectl cluster-info
kubectl get nodes
kubectl get pods              # current namespace
kubectl get pods -o wide      # node + IP
kubectl get pods -A           # all namespaces
kubectl get all               # most resources

K8s · Manage Resources

kubectl apply -f manifest.yaml    # create/update from YAML
kubectl delete -f manifest.yaml
kubectl create deployment web --image=nginx
kubectl describe pod <name>       # status + events
kubectl edit deployment web       # edit live
kubectl delete pod <name>

K8s · Scaling & Rollout

kubectl scale deployment web --replicas=3
kubectl rollout status deployment web
kubectl rollout history deployment web
kubectl rollout undo deployment web        # roll back
kubectl set image deployment/web nginx=nginx:1.27
kubectl autoscale deployment web --min=2 --max=10 --cpu-percent=80

K8s · Services & Config

kubectl get svc
kubectl expose deployment web --port=80 --type=ClusterIP
kubectl port-forward svc/web 8080:80       # tunnel to local
kubectl get ingress

kubectl create configmap cfg --from-literal=KEY=value
kubectl create secret generic db --from-literal=pass=s3cret
kubectl get configmaps ; kubectl get secrets

K8s · Namespaces & Debug

kubectl get namespaces ; kubectl create namespace dev
kubectl config get-contexts
kubectl config use-context <name>
kubectl config set-context --current --namespace=dev
